### 3.2 Phase B — Cross-Week Plan-Diversity

**File:** `packages/pipelines/src/planning/steps/select-floor-items.ts`

**Current pickWithDiversity (line ~165):**

```ts
// Existing within-plan diversity:
export async function pickWithDiversity<T extends { embedding: number[] }>(
  candidates: T[],
  count: number,
): Promise<T[]> {
  if (candidates.length === 0) return [];
  
  // Pick first by sort order (highest score)
  const picked: T[] = [candidates[0]];
  
  // For each subsequent slot: pick highest-score that maximizes min-distance from picked
  while (picked.length < count && picked.length < candidates.length) {
    const remaining = candidates.filter(c => !picked.includes(c));
    const next = remaining.reduce((best, current) => {
      const bestDist = minDistanceFromSet(best.embedding, picked.map(p => p.embedding));
      const currentDist = minDistanceFromSet(current.embedding, picked.map(p => p.embedding));
      return currentDist > bestDist ? current : best;
    });
    picked.push(next);
  }
  
  return picked;
}
```

**Patch:**

```ts
// New signature with historical seed:
export async function pickWithDiversity<T extends { embedding: number[] }>(
  candidates: T[],
  count: number,
  historicalEmbeddings: number[][] = [],  // NEW
): Promise<T[]> {
  if (candidates.length === 0) return [];
  
  // First pick: prefer candidate that's FAR from historical set
  let firstPick: T;
  if (historicalEmbeddings.length === 0) {
    firstPick = candidates[0];  // unchanged behavior when no history
  } else {
    firstPick = candidates.reduce((best, current) => {
      const bestDist = minDistanceFromSet(best.embedding, historicalEmbeddings);
      const currentDist = minDistanceFromSet(current.embedding, historicalEmbeddings);
      return currentDist > bestDist ? current : best;
    });
  }
  const picked: T[] = [firstPick];
  
  // Subsequent picks: distance from BOTH picked (within-plan) + historical (cross-week)
  while (picked.length < count && picked.length < candidates.length) {
    const remaining = candidates.filter(c => !picked.includes(c));
    const referenceSet = [
      ...picked.map(p => p.embedding),
      ...historicalEmbeddings,
    ];
    const next = remaining.reduce((best, current) => {
      const bestDist = minDistanceFromSet(best.embedding, referenceSet);
      const currentDist = minDistanceFromSet(current.embedding, referenceSet);
      return currentDist > bestDist ? current : best;
    });
    picked.push(next);
  }
  
  return picked;
}
```

// In packages/pipelines/src/planning/lib/cross-week-diversity.ts (neu)
export async function loadHistoricalPlanEmbeddings(
  projectId: string,
  lookbackWeeks: number,
): Promise<number[][]> {
  // Find last N weekly_plans for project
  const recentPlans = await db
    .select({ id: weeklyPlans.id })
    .from(weeklyPlans)
    .where(and(
      eq(weeklyPlans.projectId, projectId),
      // exclude current week — only PAST plans
      lt(weeklyPlans.isoWeek, currentIsoWeek()),
    ))
    .orderBy(desc(weeklyPlans.isoWeek))
    .limit(lookbackWeeks);
  
  if (recentPlans.length === 0) return [];
  
  // Load planned_items with brief embeddings from those plans
  const rows = await db
    .select({ embedding: topicBriefs.embedding })
    .from(plannedItems)
    .innerJoin(topicBriefs, eq(plannedItems.sourceBriefId, topicBriefs.id))
    .where(and(
      inArray(plannedItems.weeklyPlanId, recentPlans.map(p => p.id)),
      isNotNull(topicBriefs.embedding),
    ));
  
  return rows.filter(r => r.embedding !== null).map(r => r.embedding!);
}
```

**Decision-Rationale:**
- `historicalEmbeddings: number[][] = []` default — backward compatible (no history = original within-plan behavior)
- Lookback default 3 weeks — covers typical recency-bias
- Plus: env-configurable für edge-cases (long projects mit slow content rotation could go to 5+ weeks)
- Plus: helper extracted für isolated testing
- ONLY past plans — current/draft plans excluded (avoid self-referencing)

async function ensureEmbedding(brief: TopicBrief, ctx: StepContext): Promise<number[]> {
  if (brief.embedding && brief.embedding.length > 0) {
    return brief.embedding;
  }
  
  // Fallback: compute on-the-fly for legacy briefs (pre-64.15)
  log.debug({ briefId: brief.id }, "topic brief lacks embedding, computing on-the-fly");
  const embedding = await computeVoyageEmbedding(
    `${brief.topicTitle} ${brief.primaryKeyword}`,
    { ctx },
  );
  
  // Plus: lazy-backfill into DB
  await db.update(topicBriefs)
    .set({ embedding })
    .where(eq(topicBriefs.id, brief.id));
  
  return embedding;
}

// apps/api/src/scripts/backfill-brief-embeddings.ts (neu)
import { db, topicBriefs } from "@marketing-auto/db";
import { eq, isNull } from "@marketing-auto/db";
import { computeVoyageEmbedding } from "@marketing-auto/adapter-voyage";
import { parseArgs } from "node:util";

interface BackfillOptions {
  projectId?: string;
  apply: boolean;
  batchSize: number;
}

export async function backfillBriefEmbeddings(opts: BackfillOptions) {
  const where = opts.projectId
    ? and(isNull(topicBriefs.embedding), eq(topicBriefs.projectId, opts.projectId))
    : isNull(topicBriefs.embedding);
  
  const briefs = await db.select().from(topicBriefs).where(where);
  
  console.log(`Found ${briefs.length} briefs without embedding`);
  if (!opts.apply) {
    console.log("Dry-run mode — no DB writes. Pass --apply to backfill.");
    return;
  }
  
  let processed = 0;
  for (const brief of briefs) {
    try {
      const embedding = await computeVoyageEmbedding(
        `${brief.topicTitle} ${brief.primaryKeyword}`,
        { ctx: null },  // script-context, cost-tracking by project
      );
      await db.update(topicBriefs)
        .set({ embedding })
        .where(eq(topicBriefs.id, brief.id));
      processed++;
      if (processed % 10 === 0) console.log(`  ${processed}/${briefs.length}`);
    } catch (err) {
      console.error(`  Failed brief ${brief.id}:`, err);
    }
  }
  
  console.log(`Backfilled ${processed}/${briefs.length} embeddings`);
}

if (import.meta.main) {
  const { values } = parseArgs({
    args: process.argv.slice(2),
    options: {
      project: { type: "string" },
      apply: { type: "boolean", default: false },
      "batch-size": { type: "string", default: "10" },
    },
  });
  
  backfillBriefEmbeddings({
    projectId: values.project,
    apply: values.apply ?? false,
    batchSize: parseInt(values["batch-size"] ?? "10", 10),
  }).catch(err => {
    console.error("Backfill failed:", err);
    process.exit(1);
  });
}
```

- Plus: lazy-backfill in fallback path — first plan-run nach migration backfillt all briefs naturally
- Plus: explicit `apply: false` default in script (safety, Memory D20 cleanup pattern)
- Plus: per-project filter optional für targeted backfill

## 11. Memory-Update

Pattern eligible für Memory:
- **Adapter-vs-worker storage ownership for batch APIs**: when the batch lifecycle requires DB context (project_id, storagePrefix) that lives in the worker's loop, return RAW bytes from the adapter and let the worker call `convertImageToWebp`. Sync adapter still owns its R2 upload because it has direct access to the project context via `GenerateImageInput`. Pattern 119 stays consistent: every image-byte path eventually flows through the WebP adapter, but the dispatch site is whichever layer owns the per-image DB context.
- **Cross-week diversity via existing seam**: the 63.5 `initialPickedEmbeddings` field on `pickWithDiversity` was originally for cross-step Floor→Overage seeding within a plan-run. Phase B reuses it for cross-week historical seeding — same field, different semantics. Lesson: when adding a "memory-of-the-past" feature, look for an existing "seed" parameter before inventing a new one.
- **Backfill-script dry-run footgun**: an `apply=false` loop that re-reads the same SELECT each iteration spins infinitely if no UPDATE changes the predicate. Either (a) replace the loop with a `count(*)` query, or (b) paginate via OFFSET, or (c) track seen IDs. Same applies to ANY periodic-cleanup script with a "dry-run" mode that walks the predicate without mutating.
- **Test-fixture cross-tenant contamination**: `apply: true` mode in any DB-touching backfill/cleanup script must require a project-scoped filter at the API boundary, OR tests must always supply one. A test that omits the filter touches real tenant data. Flag: any new BATCH UPDATE script should add a `projectId !== null` assertion when `apply=true` (defense in depth beyond test discipline).
